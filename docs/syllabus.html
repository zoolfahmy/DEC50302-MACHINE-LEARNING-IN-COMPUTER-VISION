Topic 2: OpenCV Deep Dive

Color-Space RGB to Grayscale/HSV logic.
Contours Shape and boundary detection.
topic2_contours.py 
import cv2
import numpy as np

# 1. Load Image
image = cv2.imread('shape.jpg')
gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)

# 2. Thresholding (Binary Conversion)
ret, thresh = cv2.threshold(gray, 127, 255, 0)

# 3. Find Contours (RETR_TREE retrieves all)
contours, hierarchy = cv2.findContours(thresh, 
                                     cv2.RETR_TREE, 
                                     cv2.CHAIN_APPROX_SIMPLE)

# 4. Draw Contours (Green, Thickness 3)
cv2.drawContours(image, contours, -1, (0, 255, 0), 3)

cv2.imshow('Contours', image)
cv2.waitKey(0)
